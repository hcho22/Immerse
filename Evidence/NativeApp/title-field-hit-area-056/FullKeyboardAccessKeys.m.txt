// Throwaway host tool used for note 056's Full Keyboard Access captures; not part of any target.
// Build: clang -fobjc-arc -framework Foundation hidkeys.m -o hidkeys
// Throwaway: sends HID keyboard usages to a booted simulator through SimulatorKit, as idb does.
// usage: hidkeys <udid> <usage>[+shift] ...   (USB HID usages: Tab 43, Space 44, Return 40, Down 81, Up 82)
#import <Foundation/Foundation.h>
#import <dlfcn.h>
#import <objc/message.h>
typedef void *(*IndigoKeyFn)(uint32_t usage, int direction);
int main(int argc, char **argv) { @autoreleasepool {
    NSString *dev = @"/Applications/Xcode.app/Contents/Developer";
    dlopen("/Library/Developer/PrivateFrameworks/CoreSimulator.framework/CoreSimulator", RTLD_NOW);
    void *kit = dlopen([[dev stringByAppendingString:@"/Library/PrivateFrameworks/SimulatorKit.framework/SimulatorKit"] UTF8String], RTLD_NOW);
    if (!kit) { fprintf(stderr, "%s\n", dlerror()); return 1; }
    IndigoKeyFn key = (IndigoKeyFn)dlsym(kit, "IndigoHIDMessageForKeyboardArbitrary");
    NSError *err = nil;
    id ctx = ((id(*)(id,SEL,id,id*))objc_msgSend)(NSClassFromString(@"SimServiceContext"), NSSelectorFromString(@"sharedServiceContextForDeveloperDir:error:"), dev, &err);
    id set = ((id(*)(id,SEL,id*))objc_msgSend)(ctx, NSSelectorFromString(@"defaultDeviceSetWithError:"), &err);
    NSDictionary *byUDID = ((id(*)(id,SEL))objc_msgSend)(set, NSSelectorFromString(@"devicesByUDID"));
    id device = byUDID[[[NSUUID alloc] initWithUUIDString:@(argv[1])]];
    if (!device || !key) { fprintf(stderr, "no device or symbol\n"); return 1; }
    id hid = ((id(*)(id,SEL,id,id*))objc_msgSend)([NSClassFromString(@"SimulatorKit.SimDeviceLegacyHIDClient") alloc], NSSelectorFromString(@"initWithDevice:error:"), device, &err);
    if (!hid) { fprintf(stderr, "hid: %s\n", err.description.UTF8String); return 1; }
    dispatch_queue_t q = dispatch_queue_create("hid", 0);
    void (^send)(uint32_t, int) = ^(uint32_t usage, int dir) {
        dispatch_semaphore_t s = dispatch_semaphore_create(0);
        ((void(*)(id,SEL,void*,BOOL,id,id))objc_msgSend)(hid, NSSelectorFromString(@"sendWithMessage:freeWhenDone:completionQueue:completion:"), key(usage, dir), YES, q, ^(NSError *e){ if (e) fprintf(stderr, "send: %s\n", e.description.UTF8String); dispatch_semaphore_signal(s); });
        dispatch_semaphore_wait(s, DISPATCH_TIME_FOREVER);
        usleep(50000);
    };
    for (int i = 2; i < argc; i++) {
        NSString *a = @(argv[i]); BOOL shift = [a hasSuffix:@"+shift"];
        uint32_t u = (uint32_t)[a intValue];
        if (shift) send(225, 1);
        send(u, 1); send(u, 2);
        if (shift) send(225, 2);
        usleep(450000);
    }
} return 0; }
